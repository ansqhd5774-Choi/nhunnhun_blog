import Browserbase from '@browserbasehq/sdk';
import { chromium } from 'playwright-core';
import { readFile } from 'node:fs/promises';

const BLOG='https://nhunnhun.tistory.com';
const POST_ID='356';
const HERO='https://thumb.wikimedia.org/wikipedia/commons/thumb/0/06/Red_apple_fruits.jpg/960px-Red_apple_fruits.jpg';

for(const k of ['BROWSERBASE_API_KEY','BROWSERBASE_PROJECT_ID','BROWSERBASE_CONTEXT_ID']) if(!process.env[k]) throw new Error('E_CONFIG_'+k);

const src=JSON.parse(await readFile('posts/apple-benefits-20261002.json','utf8'));
const client=new Browserbase({apiKey:process.env.BROWSERBASE_API_KEY});
const session=await client.sessions.create({
  projectId:process.env.BROWSERBASE_PROJECT_ID,
  browserSettings:{context:{id:process.env.BROWSERBASE_CONTEXT_ID,persist:true},recordSession:false,logSession:false,solveCaptchas:false},
  timeout:300
});
let browser;
let originalMeta=null;
let updated=false;

async function jfetch(page,url,opts={}){
  return page.evaluate(async({url,opts})=>{
    const r=await fetch(url,{credentials:'include',...opts});
    const text=await r.text();
    let json=null; try{json=JSON.parse(text)}catch{}
    return {ok:r.ok,status:r.status,text,json,url:r.url};
  },{url,opts});
}

try{
  browser=await chromium.connectOverCDP(session.connectUrl);
  const context=browser.contexts()[0];
  const admin=await context.newPage();
  admin.setDefaultTimeout(30000);
  await admin.goto(BLOG+'/manage/posts',{waitUntil:'domcontentloaded'});
  if(new URL(admin.url()).origin!==BLOG) throw new Error('E_LOGIN_REQUIRED');

  const posts=await jfetch(admin,'/manage/posts.json?category=-3&page=1&searchKeyword=&searchType=title&visibility=all');
  if(!posts.ok||!posts.json) throw new Error('E_POSTS');
  const items=posts.json.items||posts.json.data?.items||posts.json.posts||[];
  originalMeta=items.find(x=>String(x.id)===POST_ID);
  if(!originalMeta) throw new Error('E_POST_META_NOT_FOUND');

  const imgRes=await fetch(HERO);
  if(!imgRes.ok) throw new Error('E_FETCH_HERO_'+imgRes.status);
  const bytes=new Uint8Array(await imgRes.arrayBuffer());
  const upload=await admin.evaluate(async({arr})=>{
    const fd=new FormData();
    fd.append('file',new Blob([new Uint8Array(arr)],{type:'image/jpeg'}),'apple-hero.jpg');
    const r=await fetch('/manage/post/attach.json',{method:'POST',credentials:'include',headers:{accept:'application/json'},body:fd});
    return {ok:r.ok,status:r.status,text:await r.text()};
  },{arr:Array.from(bytes)});
  if(!upload.ok) throw new Error('E_UPLOAD_'+upload.status);
  const up=JSON.parse(upload.text);
  if(!up.url||!up.key||!up.filename) throw new Error('E_UPLOAD_SHAPE');

  const u=new URL(up.url);
  const query=u.search.slice(1).replaceAll('&','&amp;');
  const kageRef='kage@'+up.key+'/'+up.filename+(query?'?'+query:'');
  const placeholder='[##_Image|'+kageRef+'|CDM|1.3|'+JSON.stringify({originWidth:960,originHeight:605,style:'alignCenter'})+'_##]';

  let body=src.bodyHtml;
  body=body.replace(/<p><img\b[^>]*alt=["']붉은 사과 두 개["'][^>]*><\/p>/i,placeholder);
  if(!body.includes(placeholder)) body=placeholder+body;

  const payload={
    id:POST_ID,
    title:src.title,
    content:body,
    slogan:originalMeta.slogan||'',
    visibility:20,
    category:Number(originalMeta.categoryId||0),
    tag:(src.tags||[]).join(','),
    published:1,
    password:originalMeta.postPassword||'',
    uselessMarginForEntry:1,
    cclCommercial:0,
    cclDerive:0,
    type:'post',
    attachments:[kageRef],
    thumbnail:'kage@'+up.key+'/'+up.filename,
    recaptchaValue:'',
    draftSequence:null,
    totalWritingTimeMs:0
  };

  const put=await jfetch(admin,'/manage/post/'+POST_ID+'.json',{
    method:'PUT',
    headers:{'content-type':'application/json',accept:'application/json'},
    body:JSON.stringify(payload)
  });
  if(!put.ok) throw new Error('E_UPDATE_'+put.status+'_'+put.text.slice(0,200));
  updated=true;

  await admin.waitForTimeout(2500);
  const pub=await context.newPage();
  await pub.goto(BLOG+'/'+POST_ID,{waitUntil:'domcontentloaded'});
  const verify=await pub.evaluate(()=>{
    const hero=[...document.querySelectorAll('.contents_style img')].find(x=>x.alt==='붉은 사과 두 개')||document.querySelector('.contents_style img');
    const og=document.querySelector('meta[property="og:image"]')?.content||'';
    const tw=document.querySelector('meta[property="twitter:image"]')?.content||'';
    return {hero:hero?.src||'',og,tw,title:document.title,body:(document.querySelector('.contents_style')?.innerText||'').slice(0,300)};
  });
  if(!verify.hero.includes('kakao')&&!verify.hero.includes('daumcdn')) throw new Error('E_HERO_NOT_TISTORY_CDN_'+verify.hero);
  if(/tistory_admin\/static\/images\/openGraph\/opengraph\.png/.test(verify.og)) throw new Error('E_OG_STILL_DEFAULT');
  if(!verify.title.includes('사과 효능')) throw new Error('E_TITLE_VERIFY');
  console.log('PASS_OG_IMAGE_R7 '+JSON.stringify(verify));
}catch(err){
  console.error('OG_IMAGE_R7_FAIL '+(err?.stack||err));
  throw err;
}finally{
  try{await browser?.close();}catch{}
  try{await client.sessions.update(session.id,{projectId:process.env.BROWSERBASE_PROJECT_ID,status:'REQUEST_RELEASE'});}catch{}
}
