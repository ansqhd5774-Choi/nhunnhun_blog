import Browserbase from '@browserbasehq/sdk';
import { chromium } from 'playwright-core';

for(const k of ['BROWSERBASE_API_KEY','BROWSERBASE_PROJECT_ID']){
  if(!process.env[k]) throw new Error('E_CONFIG_'+k);
}
const client=new Browserbase({apiKey:process.env.BROWSERBASE_API_KEY});
const session=await client.sessions.create({
  projectId:process.env.BROWSERBASE_PROJECT_ID,
  browserSettings:{recordSession:false,logSession:false,solveCaptchas:false},
  timeout:300
});
let browser;
try{
  browser=await chromium.connectOverCDP(session.connectUrl);
  const context=browser.contexts()[0];
  const page=await context.newPage();
  await page.setViewportSize({width:1440,height:1000});
  await page.addInitScript(()=>{
    window.__geom=[];
    window.__mut=[];
    const pack=e=>{
      if(!e) return null;
      const r=e.getBoundingClientRect(),cs=getComputedStyle(e);
      return {tag:e.tagName,id:e.id||'',cls:typeof e.className==='string'?e.className:'',
        x:r.x,y:r.y,w:r.width,h:r.height,display:cs.display,visibility:cs.visibility,
        marginTop:cs.marginTop,marginBottom:cs.marginBottom,paddingTop:cs.paddingTop,paddingBottom:cs.paddingBottom,
        fontFamily:cs.fontFamily,fontSize:cs.fontSize,lineHeight:cs.lineHeight};
    };
    const sample=()=>{
      const cw=document.querySelector('.h-entry .content-width');
      const post=document.querySelector('.e-content.post-content');
      if(cw&&post){
        window.__geom.push({
          t:performance.now(),
          post:pack(post),
          header:pack(cw.querySelector('header.hd')),
          sub:pack(cw.querySelector('header.hd .sub-info')),
          h1:pack(cw.querySelector('header.hd .hd-heading')),
          textLength:pack(cw.querySelector('.text-length')),
          popup:pack(cw.querySelector('#popupContainer')),
          children:[...cw.children].map(pack)
        });
      }
      if(performance.now()<3500) setTimeout(sample,50);
    };
    new MutationObserver(rs=>{
      for(const r of rs){
        for(const n of r.removedNodes){
          if(n.nodeType===1) window.__mut.push({t:performance.now(),kind:'remove',tag:n.tagName,id:n.id||'',cls:typeof n.className==='string'?n.className:'',parent:r.target?.className||r.target?.id||r.target?.tagName||''});
        }
        for(const n of r.addedNodes){
          if(n.nodeType===1) window.__mut.push({t:performance.now(),kind:'add',tag:n.tagName,id:n.id||'',cls:typeof n.className==='string'?n.className:'',parent:r.target?.className||r.target?.id||r.target?.tagName||''});
        }
      }
    }).observe(document,{subtree:true,childList:true});
    setTimeout(sample,0);
  });
  await page.goto('https://nhunnhun.tistory.com/356',{waitUntil:'domcontentloaded'});
  await page.waitForTimeout(4200);
  const out=await page.evaluate(()=>{
    const rows=window.__geom||[];
    const changed=[];
    let prev=null;
    for(const row of rows){
      if(!prev || !row.post || !prev.post || Math.abs(row.post.y-prev.post.y)>.5 || Math.abs(row.post.h-prev.post.h)>.5){
        changed.push(row);
      }
      prev=row;
    }
    return {changed,mutations:(window.__mut||[]).filter(x=>x.t<3500)};
  });
  console.log('DESKTOP_R6_TRACE '+JSON.stringify(out));
} finally {
  try{await browser?.close();}catch{}
  try{await client.sessions.update(session.id,{projectId:process.env.BROWSERBASE_PROJECT_ID,status:'REQUEST_RELEASE'});}catch{}
}
