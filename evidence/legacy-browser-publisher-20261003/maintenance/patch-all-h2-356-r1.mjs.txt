import Browserbase from '@browserbasehq/sdk';
import { chromium } from 'playwright-core';

const BLOG='https://nhunnhun.tistory.com', POST_ID='356';
for(const k of ['BROWSERBASE_API_KEY','BROWSERBASE_PROJECT_ID','BROWSERBASE_CONTEXT_ID']) if(!process.env[k]) throw new Error('E_CONFIG_'+k);

const client=new Browserbase({apiKey:process.env.BROWSERBASE_API_KEY});
const session=await client.sessions.create({
  projectId:process.env.BROWSERBASE_PROJECT_ID,
  browserSettings:{context:{id:process.env.BROWSERBASE_CONTEXT_ID,persist:true},recordSession:false,logSession:false,solveCaptchas:false},
  timeout:300
});
let browser;
try{
  browser=await chromium.connectOverCDP(session.connectUrl);
  const context=browser.contexts()[0], page=await context.newPage();
  page.setDefaultTimeout(30000);
  page.on('dialog',async d=>{try{await d.accept();}catch{}});
  await page.goto(BLOG+'/manage/post/'+POST_ID,{waitUntil:'domcontentloaded'});
  await page.locator('#editor-mode-layer-btn-open').click();
  await page.locator('#editor-mode-html').click();
  const cm=page.locator('.CodeMirror:visible');
  await cm.waitFor({state:'visible'});
  let html=await cm.evaluate(el=>el?.CodeMirror?.getValue?.()||'');

  // Restore original wording if the accidental copy change is still present.
  html=html.replace('1. 사과 효능, 실제로 어떤 점이 좋을까?','1. 사과를 먹을 때 기대할 수 있는 점');

  // Apply ONE consistent strong H2 visual system to every H2 without changing text.
  html=html.replace(/<h2(?:\s+[^>]*)?>/gi,'<h2 style="margin:44px 0 20px;padding:13px 16px;border-left:6px solid #2563eb;background:#f4f7fb;border-bottom:0;font-size:24px;line-height:1.4;font-weight:800;color:#111827;border-radius:0 6px 6px 0;">');

  const code=page.locator('.CodeMirror:visible .CodeMirror-code');
  await code.click();
  await page.keyboard.press('ControlOrMeta+A');
  await page.keyboard.insertText(html);
  await page.waitForTimeout(500);
  await page.locator('#publish-layer-btn').click();

  let submit=null;
  for(const name of ['수정','변경사항 저장','완료','공개 발행']){
    const b=page.getByRole('button',{name,exact:true});
    if(await b.count()){submit=b.last();break;}
  }
  if(!submit) throw new Error('E_SUBMIT');
  await submit.click();
  await page.waitForTimeout(3500);

  const pub=await context.newPage();
  await pub.goto(BLOG+'/'+POST_ID,{waitUntil:'domcontentloaded'});
  const result=await pub.evaluate(()=>{
    const hs=[...document.querySelectorAll('.contents_style h2')];
    return {
      count:hs.length,
      texts:hs.map(x=>x.textContent.trim()),
      styles:hs.map(x=>x.getAttribute('style')||'')
    };
  });
  if(result.count<10) throw new Error('E_H2_COUNT_'+result.count);
  if(!result.texts.includes('1. 사과를 먹을 때 기대할 수 있는 점')) throw new Error('E_WORDING');
  if(!result.styles.every(s=>s.includes('border-left:6px solid #2563eb')&&s.includes('background:#f4f7fb'))) throw new Error('E_STYLE_MISMATCH');
  console.log('PASS_ALL_H2_VISUAL '+JSON.stringify({count:result.count}));
}finally{
  try{await browser?.close();}catch{}
  try{await client.sessions.update(session.id,{projectId:process.env.BROWSERBASE_PROJECT_ID,status:'REQUEST_RELEASE'});}catch{}
}
